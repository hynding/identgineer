# Identity components in React

The identity elements are custom elements that take the identity client as a JavaScript property,
`client`. React needs nothing special for them, only a ref: React 18 would write an object prop as
the attribute `client="[object Object]"`, and React 19 does the same while the element is not yet
defined. Create one client for the app, hand it to every element, and read
[security.md](./security.md) before going live. `<idg-session-guard>` is a convenience for the
interface and **not access control**.

## 1. One client and the components

```ts
// identity.ts
import { createIdentityClient } from "@identgineer/identity";

export const client = createIdentityClient({ baseUrl: import.meta.env.VITE_CMS_URL });
```

```tsx
// main.tsx
import { defineCustomElements } from "@identgineer/components/loader";

void defineCustomElements();
```

Load `/tokens/index.css`, and put the pre-hydration rule from
[plain-html.md](./plain-html.md#1-load-the-tokens-and-the-components) in a stylesheet.

## 2. Types for JSX

`Components` is generated by the components package. Declare the elements you use:

```tsx
// identity-elements.d.ts
import type { Components } from "@identgineer/components";
import type { DetailedHTMLProps, HTMLAttributes } from "react";

type Element<T> = DetailedHTMLProps<HTMLAttributes<HTMLElement>, HTMLElement> & Partial<T>;

declare module "react" {
  namespace JSX {
    interface IntrinsicElements {
      "idg-sign-in": Element<Pick<Components.IdgSignIn, "providerCallback">>;
      "idg-register": Element<object>;
      "idg-password-reset": Element<Pick<Components.IdgPasswordReset, "mode">>;
      "idg-profile": Element<Pick<Components.IdgProfile, "avatarSrc">>;
      "idg-session-guard": Element<object>;
      "idg-user-menu": Element<Pick<Components.IdgUserMenu, "displayName" | "avatarSrc">>;
    }
  }
}
```

## 3. A hook that hands over the client and listens

```tsx
// use-identity-element.ts
import { useEffect, useRef } from "react";
import { client } from "./identity";

/**
 * A ref for an identity element: it gets the client and `properties` as properties, and
 * `listeners` as event listeners, removed again when the component goes.
 */
export function useIdentityElement<T extends HTMLElement>(
  properties: Record<string, unknown> = {},
  listeners: Record<string, (event: CustomEvent) => void> = {},
) {
  const ref = useRef<T>(null);
  useEffect(() => {
    const element = ref.current;
    if (!element) return;
    Object.assign(element, properties);
    (element as HTMLElement & { client?: unknown }).client = client;
    const entries = Object.entries(listeners).map(
      ([type, listener]) => [type, (event: Event) => listener(event as CustomEvent)] as const,
    );
    for (const [type, listener] of entries) element.addEventListener(type, listener);
    return () => {
      for (const [type, listener] of entries) element.removeEventListener(type, listener);
    };
  });
  return ref;
}
```

The effect sets the client on every render, which costs nothing: the element ignores a client it
already has. The other properties are set on every render too, so keep their values stable (a
module constant or `useMemo`); a new object each time makes the element draw again.

## 4. Pages

```tsx
import type { SignedInDetail } from "@identgineer/components";
import { useNavigate } from "react-router-dom";
import { useIdentityElement } from "./use-identity-element";

const SIGN_IN = { providers: [{ name: "github", label: "GitHub" }] };

export function SignInPage() {
  const navigate = useNavigate();
  const ref = useIdentityElement(SIGN_IN, {
    idgSignedIn: (event) => {
      const { user } = event.detail as SignedInDetail;
      navigate("/account", { state: { username: user.username } });
    },
  });
  return (
    <idg-sign-in ref={ref}>
      <a slot="reset" href="/reset">
        Forgot your password?
      </a>
    </idg-sign-in>
  );
}
```

```tsx
import type { SessionChangeDetail, SignedOutDetail } from "@identgineer/components";
import { useIdentityElement } from "./use-identity-element";

export function AccountPage() {
  const guard = useIdentityElement(
    {},
    {
      idgSessionChange: (event) => console.debug((event.detail as SessionChangeDetail).status),
    },
  );
  const profile = useIdentityElement();
  return (
    <idg-session-guard ref={guard}>
      <idg-profile ref={profile} />
      <p slot="fallback">
        <a href="/sign-in">Sign in</a> to see your account.
      </p>
    </idg-session-guard>
  );
}

export function UserMenu() {
  const ref = useIdentityElement(
    {},
    {
      idgSignedOut: (event) => {
        if ((event.detail as SignedOutDetail).outcome !== "superseded") location.assign("/");
      },
    },
  );
  return (
    <idg-user-menu ref={ref}>
      <a slot="signed-out" href="/sign-in">
        Sign in
      </a>
    </idg-user-menu>
  );
}
```

For the reset page, pass the code from the address as a property: `useIdentityElement({ code })`
on `<idg-password-reset mode="complete">`.

## 5. Text

`labels` is a property too. Pass a partial object of strings, which is merged over the English
defaults, and keep it stable as above:

```tsx
const GERMAN = { labels: { submit: "Anmelden", password: "Passwort" } };

const ref = useIdentityElement(GERMAN);
```

## 6. Server rendering

The elements draw nothing on the server, and the identity client needs a browser. Create the
client and call `defineCustomElements()` on the client only, for example in a `"use client"`
module in Next.js. The server's HTML then has the bare tags, which the pre-hydration rule keeps out
of sight until they hydrate.
